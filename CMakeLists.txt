cmake_minimum_required(VERSION 3.16)
project(mdread VERSION 0.1.0 LANGUAGES CXX)

if(NOT UNIX)
  message(FATAL_ERROR "mdread currently supports macOS and Linux")
endif()

find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(MD4C QUIET IMPORTED_TARGET md4c-html)
endif()
if(TARGET PkgConfig::MD4C)
  set(MDREAD_MARKDOWN_LIBRARY PkgConfig::MD4C)
else()
  find_path(MD4C_INCLUDE_DIR md4c-html.h HINTS /opt/homebrew/include /usr/local/include)
  find_library(MD4C_HTML_LIBRARY md4c-html HINTS /opt/homebrew/lib /usr/local/lib)
  find_library(MD4C_LIBRARY md4c HINTS /opt/homebrew/lib /usr/local/lib)
  if(NOT MD4C_INCLUDE_DIR OR NOT MD4C_HTML_LIBRARY OR NOT MD4C_LIBRARY)
    message(FATAL_ERROR "Install MD4C: brew install md4c (macOS) or apt install libmd4c-dev libmd4c-html0-dev (Debian/Ubuntu)")
  endif()
  add_library(mdread_markdown INTERFACE)
  target_include_directories(mdread_markdown INTERFACE "${MD4C_INCLUDE_DIR}")
  target_link_libraries(mdread_markdown INTERFACE "${MD4C_HTML_LIBRARY}" "${MD4C_LIBRARY}")
  set(MDREAD_MARKDOWN_LIBRARY mdread_markdown)
endif()

add_executable(mdread src/main.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS third_party/dompurify/purify.min.js)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/third_party/dompurify/purify.min.js" DOMPURIFY_SOURCE)
configure_file(src/purify.hpp.in generated/purify.hpp @ONLY)
include(cmake/EmbedMathAssets.cmake)
mdread_embed_math_assets("${CMAKE_CURRENT_BINARY_DIR}/generated/math_assets.hpp")
target_include_directories(mdread PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_features(mdread PRIVATE cxx_std_17)
target_compile_options(mdread PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(mdread PRIVATE ${MDREAD_MARKDOWN_LIBRARY})
find_package(Threads REQUIRED)
target_link_libraries(mdread PRIVATE Threads::Threads)
option(MDREAD_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(MDREAD_SANITIZERS)
  target_compile_options(mdread PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mdread PRIVATE -fsanitize=address,undefined)
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  target_sources(mdread PRIVATE src/window_macos.mm)
  set_source_files_properties(src/window_macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(mdread PRIVATE "-framework AppKit" "-framework WebKit")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET gtk+-3.0 webkit2gtk-4.1)
  target_sources(mdread PRIVATE src/window_linux.cpp)
  target_link_libraries(mdread PRIVATE PkgConfig::WEBKIT)
endif()
include(GNUInstallDirs)
install(TARGETS mdread RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES third_party/dompurify/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/mdread/licenses RENAME DOMPurify-LICENSE)
install(FILES third_party/katex/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/mdread/licenses RENAME KaTeX-LICENSE)
install(FILES third_party/katex/FONTS-LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/mdread/licenses RENAME KaTeX-FONTS-LICENSE)

include(CTest)
option(MDREAD_TEST_NATIVE_WINDOW "Run macOS window tests (requires a desktop session)" OFF)
if(BUILD_TESTING)
  find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
  if(Python3_FOUND)
    add_test(NAME vendored_assets COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_vendored.py")
    add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:mdread>")
    set_tests_properties(integration PROPERTIES TIMEOUT 30)
    if(APPLE)
      add_test(NAME mac_setup COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/mac_setup.py")
      set_tests_properties(mac_setup PROPERTIES TIMEOUT 60)
    endif()
    if(APPLE AND MDREAD_TEST_NATIVE_WINDOW)
      add_executable(mdread_window_test tests/window_macos.mm src/window_macos.mm)
      target_include_directories(mdread_window_test PRIVATE src)
      target_compile_options(mdread_window_test PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
      target_link_libraries(mdread_window_test PRIVATE "-framework AppKit" "-framework WebKit")
      add_test(NAME native_window COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/native_window.py"
        "$<TARGET_FILE:mdread>" "$<TARGET_FILE:mdread_window_test>")
      set_tests_properties(native_window PROPERTIES TIMEOUT 40)
      add_test(NAME math_sample COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/math_corpus.py"
        "$<TARGET_FILE:mdread>" "$<TARGET_FILE:mdread_window_test>" "${CMAKE_CURRENT_SOURCE_DIR}/examples/formulas.md")
      set_tests_properties(math_sample PROPERTIES TIMEOUT 30)
      add_test(NAME review_window COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/review_window.py"
        "$<TARGET_FILE:mdread>" "$<TARGET_FILE:mdread_window_test>")
      set_tests_properties(review_window PROPERTIES TIMEOUT 35)
    endif()
  endif()
endif()
